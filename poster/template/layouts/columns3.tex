%% Figure sizes: a share of the block's inner width (\linewidth) minus room for labels, capped at

\providecommand{\gridW}{\dimexpr(\linewidth-12\pu)/3\relax}            % block 6: 2 x 3 panels
\providecommand{\gridH}{\minsize{(\linewidth-12\pu)*19/100}{0.103\colH}}
\providecommand{\barW}{\dimexpr(\linewidth-6.5\pu)/2\relax}            % block 8: two panels
\providecommand{\barH}{\minsize{(\linewidth-6.5\pu)*33/100}{0.146\colH}}
\providecommand{\ablW}{\dimexpr(\linewidth-19\pu)/2\relax}             % block 9: two panels + row labels
