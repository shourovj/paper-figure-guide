%% venues.tex: page size, font scale and layout for each venue. Input from poster.tex after
%% \venue is set and before posterstyle.tex. Sizes are from each venue's poster page as read on
%% 2026-11-24 (GUIDELINE.md, section 14.1 lists the pages). Venues change sizes from year to
%% year: check the current page before printing, and never print larger than its maximum.
%%
%% \definevenue{key}{width cm}{height cm}{scale}{layout}{compact: true/false}
%%   scale   beamerposter font scale. Tuned so the template's nine example blocks fill about
%%           85% of every column, as on the reference NeurIPS 2026 poster (scale 1.36, body
%%           28-34 pt). Bigger posters get bigger type; compact posters trade blocks for type.
%%   layout  columns3 (landscape, width/height 1.2-1.7), columns4 (landscape, about 2:1),
%%           portrait2 (portrait)
%%   compact true drops three blocks (2, 5 and 8 in the template) so a small poster keeps
%%           readable type; the remaining blocks renumber themselves.
\newif\ifcompact
\newcommand{\definevenue}[6]{%
  \expandafter\def\csname posterpreset@#1\endcsname{%
    \def\posterwidth{#2}\def\posterheight{#3}\def\posterscale{#4}\def\posterlayout{#5}%
    \csname compact#6\endcsname}}
