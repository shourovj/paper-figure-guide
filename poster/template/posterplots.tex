%% posterplots.tex: the house figure style (GUIDELINE.md) for pgfplots at poster scale.
%% Input after pgfplots is loaded (posterstyle.tex does this). Reusable as is.
%%
%% Scale: every width, marker and gap below is given for beamerposter scale 1.36 (the NeurIPS
%% 2026 reference poster) and multiplied by \posterscale / 1.36, so a plot looks the same on an
%% A0 portrait poster at scale 1.1 as on a 213 cm CVPR poster at 1.7. Plot text follows the
%% poster's font sizes (\figtick etc.), which beamerposter already scales.
%%   - sans text (the guide's setting for posters), lowercase axis labels with units
%%   - thin full frame, no tick marks, light grid behind the data
%%   - fixed colour roles, each with a second encoding (marker, dash or hatch)
%%   - bars in light shades mixed with white (46%, ours 66%, oracle 30%), baselines hatched
%%   - one legend, ours first; text in ink, never in a series colour

\usetikzlibrary{patterns.meta}

%% ---------------- colour roles (GUIDELINE.md, section 4) ----------------
\definecolor{oursc}{HTML}{2E7BD6}     % ours: always
\definecolor{ac}{HTML}{C9302C}        % a: strongest baseline
\definecolor{bc}{HTML}{BF8700}        % b: baseline
\definecolor{cc}{HTML}{A23BA6}        % c: baseline
\definecolor{dc}{HTML}{159E86}        % d: baseline
\definecolor{basec}{HTML}{8C8C8C}     % base: untreated model, background points, connectors
\definecolor{oraclec}{HTML}{333333}   % oracle: upper bound or reference
\colorlet{variantc}{oursc!60!white}   % a variant of ours shares its hue
\definecolor{inkc}{HTML}{1A1A1A}
\definecolor{mutedc}{HTML}{666666}
\definecolor{gridc}{HTML}{E4E4E4}

%% Figure text sizes, captured here so every plot uses the same sizes even inside a column
%% that enlarges its body text with \largertext. At scale 1.36: 28.2 pt, 33.8 pt, 23.5 pt.
\let\figtick\small          % ticks, legends
\let\figlabel\normalsize    % axis labels, panel titles
\let\figvalue\footnotesize  % value labels

%% ---------------- scale ----------------
%% \pu is one "plot unit": 1 cm at scale 1.36, proportionally less or more at other scales.
%% Use it for gaps and offsets inside plots (horizontal sep=2.0\pu, xshift=-0.55\pu, ...).
\pgfmathsetmacro{\plotk}{\posterscale/1.36}
\newlength{\pu}            \pgfmathsetlength{\pu}{1cm*\plotk}
\newlength{\plotW}\newlength{\plotH}   % a plot's panel size, set by each plot macro
\newcommand{\pkdim}[2]{\newlength{#1}\pgfmathsetlength{#1}{#2*\plotk}}
\pkdim{\lwOurs}{6.3pt}     \pkdim{\mkOurs}{6.6pt}
\pkdim{\lwBase}{4.6pt}     \pkdim{\mkSq}{5.5pt}     \pkdim{\mkTri}{7.5pt}
\pkdim{\lwRef}{3.9pt}      \pkdim{\mkRef}{5.5pt}
\pkdim{\dashLong}{23pt}    \pkdim{\dashGap}{6.9pt}  \pkdim{\dashMid}{13.9pt} \pkdim{\dashDot}{4.6pt}
\pkdim{\dashRef}{15.4pt}   \pkdim{\dashRefGap}{7.7pt}
\pkdim{\hatchSep}{0.41cm}  \pkdim{\hatchLw}{1.9pt}
% patterns.meta builds a name from its options, so it needs plain values, not registers
\edef\hatchSepD{\the\hatchSep}\edef\hatchLwD{\the\hatchLw}
\pkdim{\lwDot}{4.1pt}      \pkdim{\rDot}{11pt}      \pkdim{\lwConn}{6pt}
\pkdim{\lwAxis}{2.3pt}     \pkdim{\lwGrid}{1.9pt}

%% ---------------- series styles (TikZ, so they also work in legends) ----------------
\tikzset{
  s ours/.style={oursc, line width=\lwOurs, mark=*, mark size=\mkOurs, mark options={solid}},
  s a/.style={ac, line width=\lwBase, mark=square*, mark size=\mkSq, mark options={solid}},
  s b/.style={bc, line width=\lwBase, dash pattern=on \dashLong off \dashGap,
    mark=triangle*, mark size=\mkTri, mark options={solid}},
  s c/.style={cc, line width=\lwBase, dash pattern=on \dashMid off \dashDot on \dashDot off \dashDot,
    mark=diamond*, mark size=\mkTri, mark options={solid}},
  s d/.style={dc, line width=\lwBase, dash pattern=on \dashDot off \dashDot,
    mark=triangle*, mark size=\mkTri, mark options={solid, rotate=180}},
  s base/.style={basec, line width=\lwRef, mark=*, mark size=\mkRef, mark options={solid}},
  s oracle/.style={oraclec, line width=\lwRef, dash pattern=on \dashRef off \dashRefGap, mark=none},
  s variant/.style={variantc, line width=\lwBase, dash pattern=on \dashLong off \dashGap, mark=none},
  % bars: light shade mixed with white; baselines keep a hatch in the full role colour
  hatch/.style 2 args={postaction={pattern={Lines[angle=#2, distance=\hatchSepD, line width=\hatchLwD]}, pattern color=#1}},
  b ours/.style={fill=oursc!66!white, draw=none},
  b a/.style={fill=ac!46!white, draw=none, hatch={ac}{45}},
  b b/.style={fill=bc!46!white, draw=none, hatch={bc}{-45}},
  b c/.style={fill=cc!46!white, draw=none,
    postaction={pattern={Dots[distance=\hatchSepD, radius=\hatchLwD]}, pattern color=cc}},
  b d/.style={fill=dc!46!white, draw=none,
    postaction={pattern={Hatch[distance=\hatchSepD, line width=\hatchLwD]}, pattern color=dc}},
  b base/.style={fill=basec!46!white, draw=none},
  b oracle/.style={fill=oraclec!30!white, draw=none},   % reference bars lighter, so ours reads first
  b variant/.style={fill=oursc!40!white, draw=none},
  % dumbbell dots: hollow = before (e.g. peak), filled = after (e.g. final)
  d before/.style={draw=#1, fill=white, line width=\lwDot},
  d after/.style={draw=#1, fill=#1, line width=\lwDot},
}

%% ---------------- legends and keys (set in the figure font) ----------------
% \legline{style}{label}  line sample      \legbar{style}{label}  bar swatch
% \legdot{style}{label}   dot              \figlegend{entries}    one centred row under a figure
\newcommand{\legline}[2]{\mbox{\tikz[baseline=-0.6ex]\draw[#1, mark indices={2}] plot coordinates {(0,0) (1.1\pu,0) (2.2\pu,0)};\hspace{0.4em}#2}}
\newcommand{\legbar}[2]{\mbox{\tikz[baseline=-0.1ex]\path[#1] (0,-0.3\pu) rectangle (1.0\pu,0.5\pu);\hspace{0.4em}#2}}
\newcommand{\legdot}[2]{\mbox{\tikz[baseline=-0.6ex]\path[#1] (0,0) circle (0.8\rDot);\hspace{0.4em}#2}}
\newcommand{\figlegend}[1]{{\centering\sffamily\figtick\color{inkc}#1\par}}
% A text line that starts with a series' line sample. Lines like this under a plot can be its
% legend: \keyline{s a}{\textbf{Baseline A:} what happens to it.}
\newcommand{\legkey}[1]{\tikz[baseline=-0.6ex]\draw[#1, mark indices={2}] plot coordinates {(0,0) (0.9\pu,0) (1.8\pu,0)};}
\newcommand{\keyline}[2]{{\hangindent=2.5\pu\hangafter=1\noindent\makebox[2.5\pu][l]{\legkey{#1}}#2\par}\vspace{0.2\pu}}

%% ---------------- axis style ----------------
\pgfplotsset{
  posteraxis/.style={
    scale only axis,
    axis line style={draw=inkc, line width=\lwAxis},
    xmajorgrids, ymajorgrids, grid style={gridc, line width=\lwGrid},
    tick style={draw=none},
    /pgf/number format/assume math mode=true,   % sans digits
    tick label style={font=\sffamily\figtick, inkc},
    label style={font=\sffamily\figlabel, inkc},
    title style={font=\sffamily\figlabel, inkc, yshift=-0.2\pu},
    every node near coord/.append style={font=\sffamily\figvalue, inkc},
  },
  % bar charts: no grid along the categorical axis
  posterbars/.style={posteraxis, xmajorgrids=false},
}

%% Second-level category label under a group of ticks, e.g. the model family under its sizes.
%% Use inside an axis with clip=false: \famlabel{symbolic x coord}{text}
\newcommand{\famlabel}[2]{\node[font=\sffamily\figtick, text=inkc, anchor=north, yshift=-1.3\pu] at (axis cs:#1,0) {#2};}

%% Dumbbell row inside an axis: \Dumbbell{colour}{row}{before}{after}
\newcommand{\Dumbbell}[4]{%
  \draw[basec, line width=\lwConn] (axis cs:#4,#2) -- (axis cs:#3,#2);
  \path[d before=#1] (axis cs:#3,#2) circle (\rDot);
  \path[d after=#1]  (axis cs:#4,#2) circle (\rDot);
}
